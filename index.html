Architecture

How the engine is built

Three build stages, each one runnable and complete on its own before the next was layered on.

01 · ECE PHYSICS

Single LIF neuron

A pure-Python LIFNeuron class modeling the membrane as an RC circuit, integrated with Euler's method. Charges, spikes at threshold, resets, leaks back to rest.

02 · MATRIX MATH

1,000-neuron network

Rewritten in NumPy: a spike vector times an N×N synaptic weight matrix replaces thousands of nested loops, computing every neuron's incoming current at once.

03 · BROWSER BRIDGE

WASM + canvas render

The same class runs unmodified inside PyScript. A JS loop drives network.step() at 30fps and paints every neuron's voltage straight to canvas.

packages = ["numpy"] import numpy as np from pyscript import window from pyscript.ffi import to_js class LIFNetwork: def __init__(self, num_neurons=150): self.N = num_neurons self.tau = 15.0 self.R = 2.0 self.v_rest = 0.0 self.v_threshold = 1.0 self.v_reset = 0.0 self.v = np.full(self.N, self.v_rest) self.spikes = np.zeros(self.N, dtype=bool) self.W = np.random.uniform(0, 0.08, (self.N, self.N)) np.fill_diagonal(self.W, 0) self.x_coords = np.random.uniform(20, 580, self.N) self.y_coords = np.random.uniform(20, 380, self.N) def step(self, external_current): synaptic_current = np.dot(self.W, self.spikes) total_current = external_current + synaptic_current dv = (1.0 / self.tau) * (-(self.v - self.v_rest) + (self.R * total_current)) self.v += dv self.spikes = self.v >= self.v_threshold self.v[self.spikes] = self.v_reset return self.v, self.spikes network = LIFNetwork(150) time_step = 0 trigger_injection = False def get_network_topology(): return to_js(network.x_coords), to_js(network.y_coords), to_js(network.W) window.get_network_topology = get_network_topology def trigger_current(): global trigger_injection trigger_injection = True window.trigger_current = trigger_current def run_simulation_step(): global time_step, trigger_injection ext_input = np.zeros(network.N) if trigger_injection: ext_input[0:15] = 2.0 trigger_injection = False voltages, spikes = network.step(ext_input) time_step += 1 v_js = to_js(voltages) s_js = to_js(spikes) return time_step, v_js, s_js window.run_simulation_step = run_simulation_step
